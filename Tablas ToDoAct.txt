CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) UNIQUE NOT NULL,
    contraseña VARCHAR(50) UNIQUE NOT NULL,
    nombrecompleto VARCHAR(255) NOT NULL,
    fechanacimiento DATE NOT NULL,
    genero VARCHAR(20) NOT NULL,
    estilo_aprendizaje VARCHAR(30)
);


CREATE TABLE contenidos (
    id SERIAL PRIMARY KEY,
    area VARCHAR(50) NOT NULL,
    slug VARCHAR(100) NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    orden INTEGER DEFAULT 0,
    activo BOOLEAN DEFAULT true,
    UNIQUE (area, slug)
);


CREATE TABLE niveles_contenido (
    id SERIAL PRIMARY KEY,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20) NOT NULL,
    UNIQUE (contenido_id, nivel)
);


CREATE TABLE progreso_contenido_usuario (
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    diagnostico_realizado BOOLEAN DEFAULT false,
    nivel_asignado VARCHAR(20),
    contenido_completado BOOLEAN DEFAULT false,
    tiempo_total_diagnostico INTEGER,
    fecha_inicio_diagnostico TIMESTAMP,
    fecha_fin_diagnostico TIMESTAMP,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (usuario, contenido_id)
);


CREATE TABLE contenidos_archivos (
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20) NOT NULL,
    estilo VARCHAR(30) NOT NULL,
    archivo TEXT NOT NULL,
    PRIMARY KEY (contenido_id, nivel, estilo)
);


CREATE TABLE examenes_diagnostico (
    id SERIAL PRIMARY KEY,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    activo BOOLEAN DEFAULT true,
    UNIQUE (contenido_id)
);


CREATE TABLE preguntas_diagnostico (
    id SERIAL PRIMARY KEY,
    examen_id INTEGER NOT NULL REFERENCES examenes_diagnostico(id) ON DELETE CASCADE,
    pregunta TEXT NOT NULL,
    opcion_a TEXT NOT NULL,
    opcion_b TEXT NOT NULL,
    opcion_c TEXT NOT NULL,
    opcion_d TEXT NOT NULL,
    respuesta_correcta CHAR(1) NOT NULL
);


CREATE TABLE respuestas_diagnostico (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    pregunta_id INTEGER NOT NULL REFERENCES preguntas_diagnostico(id) ON DELETE CASCADE,
    respuesta_dada CHAR(1) NOT NULL,
    es_correcta BOOLEAN NOT NULL,
    intento INTEGER NOT NULL DEFAULT 1,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE resultados_quiz (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20) NOT NULL,
    estilo VARCHAR(30) NOT NULL,
    aciertos INTEGER NOT NULL DEFAULT 0,
    total_preguntas INTEGER NOT NULL DEFAULT 5,
    puntaje INTEGER NOT NULL DEFAULT 0,
    tiempo_total INTEGER,
    intento INTEGER NOT NULL DEFAULT 1,
    aprobado BOOLEAN NOT NULL DEFAULT false,
    fecha_inicio TIMESTAMP,
    fecha_fin TIMESTAMP,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    errores INTEGER NOT NULL DEFAULT 0,
    tipo_resultado VARCHAR(20) NOT NULL DEFAULT 'quiz',
    nivel_resultante VARCHAR(20),
    session_uuid VARCHAR(64)
);


CREATE TABLE resultados_reforzamiento (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20) NOT NULL,
    estilo VARCHAR(30) NOT NULL,
    pagina INTEGER NOT NULL,
    apartado_clave VARCHAR(100) NOT NULL,
    tipo_actividad VARCHAR(50) NOT NULL,
    aciertos INTEGER NOT NULL DEFAULT 0,
    errores INTEGER NOT NULL DEFAULT 0,
    intento INTEGER NOT NULL DEFAULT 1,
    completado BOOLEAN NOT NULL DEFAULT true,
    detalle_json JSONB NOT NULL DEFAULT '{}'::jsonb,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    session_uuid VARCHAR(64)
);


CREATE TABLE sesiones_reforzamiento (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20) NOT NULL,
    estilo VARCHAR(30) NOT NULL,
    tiempo_total INTEGER NOT NULL DEFAULT 0,
    intento INTEGER NOT NULL DEFAULT 1,
    evento_cierre VARCHAR(30) NOT NULL DEFAULT 'quiz',
    fecha_inicio TIMESTAMP,
    fecha_fin TIMESTAMP,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    session_uuid VARCHAR(64)
);


CREATE TABLE resultados_vark (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    usuario_id INTEGER REFERENCES usuarios(id) ON DELETE SET NULL,
    intento INTEGER NOT NULL DEFAULT 1,
    respuestas_json JSONB NOT NULL DEFAULT '{}'::jsonb,
    totales_v INTEGER NOT NULL DEFAULT 0,
    totales_a INTEGER NOT NULL DEFAULT 0,
    totales_r INTEGER NOT NULL DEFAULT 0,
    totales_k INTEGER NOT NULL DEFAULT 0,
    codigo_estilo VARCHAR(1),
    estilo_resultado VARCHAR(30),
    es_clasificable BOOLEAN NOT NULL DEFAULT false,
    motivo_no_clasificable VARCHAR(40),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE inferencias_bayes (
    id SERIAL PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL,
    contenido_id INTEGER NOT NULL REFERENCES contenidos(id) ON DELETE CASCADE,
    nivel VARCHAR(20),
    estilo VARCHAR(30),
    origen VARCHAR(20) NOT NULL,
    referencia_id INTEGER,
    session_uuid VARCHAR(64),
    recomendacion VARCHAR(30) NOT NULL,
    confianza NUMERIC(10,6) NOT NULL DEFAULT 0,
    probabilidades_json JSONB NOT NULL DEFAULT '[]'::jsonb,
    rasgos_json JSONB NOT NULL DEFAULT '[]'::jsonb,
    modelo_version VARCHAR(40) NOT NULL,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE puntuaciones (
    id SERIAL PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    id_actividad INTEGER,
    categoria VARCHAR(255),
    nivel VARCHAR(255),
    puntuacion INTEGER NOT NULL DEFAULT 0,
    tiempo INTEGER NOT NULL DEFAULT 0,
    errores INTEGER NOT NULL DEFAULT 0,
    estilo_mostrado VARCHAR(255),
    fecha TIMESTAMP DEFAULT now()
);
